"""Opt-in, bounded live measurement through the unchanged hosted answer function."""
from __future__ import annotations

import argparse
from datetime import datetime, timezone
import importlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import time
from unittest.mock import patch

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from evaluation.run import load_dataset, sha256


def score_response(case, result, chunks):
    """Transparent fixture checks; this is not a general semantic accuracy score."""
    allowed = {f"[Source: {c['metadata']['doc_title']}, Section: {c['metadata']['section_title']}]" for c in chunks}
    citations = result.get('citations', [])
    refused = result['refused']
    facts = [bool(re.search(pattern, result['answer'], re.I)) for pattern in case.get('required_fact_patterns', [])]
    valid_labels = bool(citations) and all(label in allowed for label in citations)
    correct_refusal = refused == (not case['answerable'])
    facts_pass = all(facts) if case['answerable'] else None
    return {
        'expected_answerable': case['answerable'],
        'refused': refused,
        'refusal_decision_matches_fixture': correct_refusal,
        'non_refusal_has_valid_retrieved_labels': valid_labels if not refused else None,
        'required_fact_pattern_matches': facts,
        'all_required_fact_patterns_match': facts_pass,
        'fixture_checks_pass': correct_refusal and (refused or (valid_labels and facts_pass)),
    }


def usage_fields(response):
    usage = response.get('usage') or {}
    def integer(value):
        return value if type(value) is int and value >= 0 else None
    return {
        'prompt_tokens': integer(usage.get('prompt_tokens')),
        'completion_tokens': integer(usage.get('completion_tokens')),
        'total_tokens': integer(usage.get('total_tokens')),
        'cached_prompt_tokens': integer((usage.get('prompt_tokens_details') or {}).get('cached_tokens')),
        'reasoning_tokens': integer((usage.get('completion_tokens_details') or {}).get('reasoning_tokens')),
    }


def validate_budget(value):
    if not 1 <= value <= 30:
        raise ValueError('Request budget must be between 1 and 30.')
    return value


def summarize(records):
    complete = [r for r in records if r['status'] == 'completed']
    supported = [r for r in complete if r['checks']['expected_answerable']]
    unsupported = [r for r in complete if not r['checks']['expected_answerable']]
    usage = [r['usage'] for r in records if r.get('usage')]
    return {
        'attempted_cases': len(records), 'completed_cases': len(complete),
        'error_cases': len(records) - len(complete),
        'fixture_checks_passed': sum(r['checks']['fixture_checks_pass'] for r in complete),
        'answerable_cases_completed': len(supported),
        'answerable_non_refusals': sum(not r['checks']['refused'] for r in supported),
        'answerable_required_fact_checks_passed': sum(r['checks']['all_required_fact_patterns_match'] for r in supported),
        'unsupported_cases_completed': len(unsupported),
        'unsupported_refusals': sum(r['checks']['refused'] for r in unsupported),
        'records_with_usage': len(usage),
        'token_totals': {key: sum(u[key] for u in usage if u[key] is not None) for key in ('prompt_tokens', 'completion_tokens', 'total_tokens')},
        'currency_cost': None,
    }


def save_report(path, report):
    report['summary'] = summarize(report['cases'])
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(path.suffix + '.tmp')
    temporary.write_text(json.dumps(report, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
    temporary.replace(path)


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--live', action='store_true', help='Explicitly permit paid generation requests.')
    parser.add_argument('--max-requests', type=int, default=18)
    parser.add_argument('--env-file', type=Path, help='Optional private configuration file; values are never logged.')
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args(argv)
    if not args.live:
        parser.error('No requests sent. Pass --live to permit the bounded provider run.')
    try:
        validate_budget(args.max_requests)
    except ValueError as error:
        parser.error(str(error))
    dataset, _ = load_dataset()
    cases = [case for case in dataset['cases'] if case.get('live_sample')]
    if len(cases) > args.max_requests:
        parser.error('The frozen sample exceeds the request budget; no requests sent.')
    if not cases:
        parser.error('No frozen sample cases selected; no requests sent.')
    for case in cases:
        if case['answerable'] and not case.get('required_fact_patterns'):
            parser.error('Each answerable sample requires pre-authored fact patterns; no requests sent.')
        for pattern in case.get('required_fact_patterns', []):
            re.compile(pattern, re.I)
    if args.env_file:
        from dotenv import dotenv_values
        values = dotenv_values(args.env_file)
        for name in ('OPENAI_API_KEY', 'OPENAI_MODEL', 'OPENAI_MAX_COMPLETION_TOKENS'):
            if values.get(name) and not os.getenv(name):
                os.environ[name] = values[name]
    if not os.getenv('OPENAI_API_KEY', '').strip():
        parser.error('OPENAI_API_KEY is required; no requests sent.')
    sys.path.insert(0, str(ROOT / 'web'))
    site = importlib.import_module('app')
    index = site.get_index()
    source = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
    report = {
        'schema_version': 1, 'recorded_at_utc': datetime.now(timezone.utc).isoformat(),
        'scope': 'Real local ONNX retrieval plus unchanged production answer() function; not hosted HTTP, admission, load, production-user or general semantic-quality measurement.',
        'dataset_id': dataset['dataset_id'], 'dataset_sha256': sha256(ROOT / 'evaluation/cases.json'),
        'corpus_sha256': sha256(ROOT / 'web/data/corpus.json'), 'source_base_commit': source,
        'source_sha256': {name: sha256(ROOT / name) for name in ('web/app.py', 'web/engine.py', 'evaluation/evaluate_answers.py')},
        'requested_model': os.getenv('OPENAI_MODEL', 'gpt-6-luna'),
        'max_completion_tokens': int(os.getenv('OPENAI_MAX_COMPLETION_TOKENS', '2048')),
        'request_budget': args.max_requests, 'provider_requests': 0, 'cases': [],
        'grading': 'Exact refusal state, retrieved citation-label membership and pre-authored regex fact proxies; not human assessment or factual-entailment evaluation.',
    }
    original = site.post_json
    for case in cases:
        started = time.perf_counter()
        retrieved = index.search(case['question'])
        retrieval_ms = (time.perf_counter() - started) * 1000
        captured = {}
        def measured_post(url, payload, token, timeout=15):
            if url != 'https://api.openai.com/v1/chat/completions':
                raise RuntimeError('Unexpected provider destination.')
            if report['provider_requests'] >= args.max_requests:
                raise RuntimeError('Provider request budget exhausted.')
            report['provider_requests'] += 1
            before = time.perf_counter()
            try:
                response = original(url, payload, token, timeout)
                captured.update(usage=usage_fields(response), model=response.get('model'))
                choice = (response.get('choices') or [{}])[0]
                captured['provider_output'] = (choice.get('message') or {}).get('content')
                captured['finish_reason'] = choice.get('finish_reason')
                return response
            finally:
                captured['provider_elapsed_ms'] = round((time.perf_counter() - before) * 1000, 3)
        row = {'id': case['id'], 'question': case['question'], 'retrieval_ms': round(retrieval_ms, 3)}
        try:
            with patch.object(site, 'post_json', measured_post):
                result = site.answer(case['question'], retrieved)
            row.update(status='completed', result=result, checks=score_response(case, result, retrieved))
        except Exception as error:
            row.update(status='error', error_type=type(error).__name__)
            if isinstance(getattr(error, 'status_code', None), int):
                row['http_status'] = error.status_code
        row.update(captured)
        row['elapsed_ms'] = round((time.perf_counter() - started) * 1000, 3)
        row['retrieved_sources'] = [{'doc': c['metadata']['doc_title'], 'section': c['metadata']['section_title'], 'chunk_index': c['metadata']['chunk_index']} for c in retrieved]
        report['cases'].append(row)
        save_report(args.output, report)
        print(f"{case['id']}: {row['status']} ({report['provider_requests']}/{args.max_requests} provider requests)", flush=True)
        # Stop on provider/auth/network failures; preserve schema/guard failures as observations.
        if row['status'] == 'error' and row.get('http_status') != 502:
            return 1
    return 0 if not report['summary']['error_cases'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
